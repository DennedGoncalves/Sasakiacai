🔗 Banco de dados (Supabase)
Conecte ao Supabase para persistir dados reais. Crie um projeto em supabase.com, rode o SQL abaixo e cole as credenciais.
CREATE TABLE IF NOT EXISTS produtos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nome text NOT NULL, preco numeric(10,2) DEFAULT 0, cat text, emoji text, ativo boolean DEFAULT true, est integer DEFAULT 0, bc text, badge text, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS adicionais ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nome text NOT NULL, preco numeric(10,2) DEFAULT 0, emoji text, ativo boolean DEFAULT true, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS pedidos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, comanda_num integer, cliente text DEFAULT 'Consumidor Final', itens jsonb, total numeric(10,2) DEFAULT 0, status text DEFAULT 'pendente', endereco text, pgto text, entregador text, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS financas ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, tipo text, descricao text, val numeric(10,2) DEFAULT 0, cat text, data date DEFAULT current_date, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS creditos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nome text NOT NULL, telefone text, limite numeric(10,2) DEFAULT 0, divida numeric(10,2) DEFAULT 0, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS entregadores ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nome text NOT NULL, telefone text, status text DEFAULT 'ativo', created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS caixas ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, operador text, abertura timestamptz DEFAULT now(), fechamento timestamptz, valor_inicial numeric(10,2) DEFAULT 0, vendas jsonb, total numeric(10,2) DEFAULT 0, status text DEFAULT 'aberto', created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS usuarios ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nome text NOT NULL, email text, perfil text DEFAULT 'atendente', ativo boolean DEFAULT true, created_at timestamptz DEFAULT now() );
